```python
#!/usr/bin/env python3
"""S0.1a: инвентаризация образов-попыток владельца.

Обходит BIOSDIR, корень KOT (v011-эра попытки лежат там — выяснено при
исполнении 2026-09-05) и $KOT/HNX99TF_200525_original/, для каждого файла
>=1MB классифицирует базу (минимальный суммарный дифф против V011 и LIVE),
считает регионы отличий. Вывод — TSV. Строки самой базы V011 (self, дифф 0)
и несводимых доноров (base `-`) — норма.
"""
import hashlib, os, sys, time

BIOSDIR = '/var/home/dsevosty/git/BIOS'
KOT_ROOT = '/var/home/dsevosty/downloads/bios/HNX99TF_200525_kot'
KOT_ORIG = KOT_ROOT + '/HNX99TF_200525_original'
BASES = {
    'v011': '/var/home/dsevosty/downloads/bios/HNX99TF_200525_kot/HNX99TF_200525_kot_v011_5B7C6A7F.bin',
    'E5C88C6F': '/var/home/dsevosty/git/IMPLEMENTATION/UEFIPatcher/refs/fw/HNX99TF_200525_original_E5C88C6F.bin',
}

def diff_regions(a, b):
    if len(a) != len(b):
        return None
    regions, i, n = [], 0, 0
    while i < len(a):
        if a[i] != b[i]:
            j = i
            while j < len(a) and a[j] != b[j]:
                j += 1
            regions.append((i, j - i)); n += j - i; i = j
        else:
            i += 1
    return regions, n

def main():
    rows = []
    for root in (BIOSDIR, KOT_ORIG, KOT_ROOT):
        for name in sorted(os.listdir(root)):
            p = os.path.join(root, name)
            if not os.path.isfile(p) or os.path.getsize(p) < 1_000_000:
                continue
            d = open(p, 'rb').read()
            best = ('-', 0, 0, 1 << 62)
            for bname, bpath in BASES.items():
                r = diff_regions(d, open(bpath, 'rb').read())
                if r is None:
                    continue
                regions, total = r
                if total < best[3]:
                    best = (bname, regions[0][0] if regions else -1, len(regions), total)
            rows.append((p, len(d), time.strftime('%Y-%m-%d', time.localtime(os.path.getmtime(p)))) + best)
    for r in sorted(rows, key=lambda x: (x[3], x[0])):
        print('\t'.join(map(str, r)))

main()
```

Примечание к `best`: `regions[0][0]` — офсет первого региона. Для
несжатых 16МБ-образов дифф-регионы включают перестроенные FFS-заголовки —
это нормально, классификация по минимуму суммарного диффа устойчива.

Формат FV-заголовка (PI 1.8): `ZeroVector[16] FileSystemGuid[16] FvLength:u64@0x20 Signature'_FVH'@0x28 Attributes:u32@0x2C HeaderLength:u16@0x30 Checksum:u16@0x32 ExtHeaderOffset:u16@0x34 Reserved:u8 Revision:u8`, далее BlockMap `(NumBlocks:u32, Length:u32)*` до `(0,0)`. FFS-файл (UEFITool `common/ffs.h` EFI_FFS_FILE_HEADER): `Name:GUID[16] IntegrityCheck.Header:u8@0x10 IntegrityCheck.File:u8@0x11 Type:u8@0x12 Attributes:u8@0x13 Size:u24@0x14 State:u8@0x17` (24 Б), следующий файл = `align8(start+Size)`; `0xFF`-заполнение = свободный хвост.

Скрипт (docstring-шапка по образцу `hack/hii_probe.py`: назначение, ссылка на спеку §3 S0.1, ограничение «raw-скан без LZMA-распаковки; вложенные FV внутри спанов уже найденных FV пропускаются (claimed-spans), а не помечаются»):

```python
#!/usr/bin/env python3
"""FV audit: карта firmware volumes образа + свободное место.

Разведка S0.1b (спека 2026-09-05-serial-console-ladder-design.md §3):
FV-карта живого образа, свободный хвост каждого FV, носители вставки
прошлых попыток. Не часть движка; запуск вручную:

    python3 hack/fv_audit.py <image.bin> [--json]

Ограничения: raw-скан без LZMA-распаковки (вложенные в LZMA-секции
FV не видны); FV внутри спанов уже найденных FV пропускаются
(claimed-spans); FFS-цепочка читается по заголовкам без валидации
типов секций.
"""
import json, struct, sys, uuid

def u16(d, o): return struct.unpack_from('<H', d, o)[0]
def u32(d, o): return struct.unpack_from('<I', d, o)[0]
def u64(d, o): return struct.unpack_from('<Q', d, o)[0]
def g(b): return str(uuid.UUID(bytes_le=b)).upper()

def walk_ffs(data, fv_off, hdr_len, fv_len):
    files, pad, p, end = 0, 0, fv_off + hdr_len, fv_off + fv_len
    last = p
    while p + 24 <= end:
        if data[p + 0x12] == 0xFF and data[p + 0x14] == 0xFF:
            break
        size = data[p + 0x14] | (data[p + 0x15] << 8) | (data[p + 0x16] << 16)
        if size == 0xFFFFFF or size < 24:
            break
        files += 1
        if data[p + 0x12] == 0xF0:
            pad += 1
        last = p + size
        p = (last + 7) & ~7
    return files, pad, last, end - max(last, fv_off + hdr_len)

def walk_fvs(data):
    out, claimed, pos = [], [], 0
    while True:
        i = data.find(b'_FVH', pos)
        if i < 0:
            break
        pos = i + 4
        fv = i - 0x28
        if fv < 0 or any(a <= fv < b for a, b in claimed):
            continue
        length, hlen = u64(data, fv + 0x20), u16(data, fv + 0x30)
        if length == 0 or fv + length > len(data) or hlen < 0x38:
            continue
        files, pad, last, free_tail = walk_ffs(data, fv, hlen, length)
        claimed.append((fv, fv + length))
        out.append({'offset': fv, 'guid': g(data[fv + 0x10:fv + 0x20]),
                    'length': length, 'header_len': hlen, 'files': files,
                    'pad_files': pad, 'used': last - fv, 'free_tail': free_tail})
    return out

if __name__ == '__main__':
    img = open(sys.argv[1], 'rb').read()
    fvs = walk_fvs(img)
    if '--json' in sys.argv:
        print(json.dumps(fvs, indent=1))
    else:
```

```python
#!/usr/bin/env python3
"""S0.2a: донорский $SPF — где контейнер, как устроены serial-страницы."""
import lzma, struct, sys, uuid

LZMA_GUIDS = [bytes.fromhex(h) for h in (
    '98584eee143959429d6edc7bd79403cf', '235ed80e53f23f41a03c901987b04397',
    'ea2199bd91ed4a408b2fb4d724747c8c', 'bde62ad45213fb4b909aca72a6eae889')]

def decompress(comp):
    return lzma.LZMADecompressor(format=lzma.FORMAT_ALONE).decompress(comp)

def lzma_sections(data):
    for g in LZMA_GUIDS:
        pos = 0
        while True:
            i = data.find(g, pos)
            if i < 0:
                break
            pos = i + 1
            dataoff = struct.unpack_from('<H', data, i + 16)[0]
            try:
                yield i, decompress(data[i - 4 + dataoff:])
            except lzma.LZMAError:
                pass

def u16(d, o): return struct.unpack_from('<H', d, o)[0]
def u32(d, o): return struct.unpack_from('<I', d, o)[0]

def pages(c):
    n = u32(c, 0x60)
    ptrs = [u32(c, 0x64 + 4 * k) for k in range(n)]
    for k, p in enumerate(ptrs):
        if not p or p + 0x24 > len(c):
            continue
        yield k, p, u16(c, p + 0xA), u16(c, p + 0xE)

if __name__ == '__main__':
    data = open(sys.argv[1], 'rb').read()
    tag = sys.argv[2]
    for off, dec in lzma_sections(data):
        j = dec.find(b'$SPF')
        if j < 0:
            continue
        c = dec[j:]
        open(f'/tmp/serial-s0/spf_{tag}.bin', 'wb').write(c)
        print(f'{tag}: lzma@{off:#x} spf_len~{len(c):#x} pages={u32(c, 0x60)}')
        for k, p, fid, title in pages(c):
            print(f'  p{k:03d} @{p:#07x} form={fid:#06x} title={title:#06x}')
```
